import argparse

import vertexai
from vertexai.preview.vision_models import ImageGenerationModel

def generate_bouquet_image(
    project_id: str, location: str, output_file: str, prompt: str
) -> vertexai.preview.vision_models.ImageGenerationResponse:
    """Generate an image using a text prompt.
    Args:
      project_id: Google Cloud project ID, used to initialize Vertex AI.
      location: Google Cloud region, used to initialize Vertex AI.
      output_file: Local path to the output image file.
      prompt: The text prompt describing what you want to see."""

    vertexai.init(project=project_id, location=location)

    model = ImageGenerationModel.from_pretrained("imagen-3.0-generate-002")

    images = model.generate_images(
        prompt=prompt,
        # Optional parameters
        number_of_images=1,
        seed=1,
        add_watermark=False,
    )

    images[0].save(location=output_file)

    return images

    # Load Gemini multimodal model
    multimodal_model = GenerativeModel("gemini-2.0-flash-001")

    # Load the image and create the input parts
    response = multimodal_model.generate_content(
        [
            Part.from_image(Image.load_from_file("bouquet.jpeg")),
            "Generate birthday wishes based on the image"

    )

import vertexai
from vertexai.generative_models import GenerativeModel, Part, Image
import os

def analyze_bouquet_image(project_id: str, location: str):
    """Analyze bouquet image and generate birthday wishes.

    Args:
        project_id: Google Cloud project ID
        location: Google Cloud region"""

    # Initialize Vertex AI
    vertexai.init(project=project_id, location=location)

    # Load the Gemini multimodal model (version 2.0 flash)
    model = GenerativeModel("gemini-2.0-flash-001")

    # Load the image from file
    image_path = "bouquet.jpeg"
    image_part = Part.from_image(Image.load_from_file(image_path))

    # Generate birthday wishes based on the image
    print("Generate the birthday wishe based on your image...")
    response_stream = model.generate_content(
        [image_part, Part.from_text("Analyze this bouquet image and create birthday wishes in English.")],
        stream=True
    )

    full_response = ""
    for chunk in response_stream:
        if chunk.text:
            print(chunk.text, end="", flush=True)
            full_response += chunk.text
    print("\n")

    print("✅ Birthday wishes generation completed!")
    return full_response

# Execute the function

import vertexai
from vertexai.generative_models import GenerativeModel, Part, Image
import os

def analyze_bouquet_image(project_id: str, location: str):
    """Analyze bouquet image and generate birthday wishes.

    Args:
        project_id: Google Cloud project ID
        location: Google Cloud region"""

    # Initialize Vertex AI
    vertexai.init(project=project_id, location=location)

    # Load the Gemini multimodal model (version 2.0 flash)
    model = GenerativeModel("gemini-2.0-flash-001")

    # Load the image from file
    image_path = "generated_bouquet.png"
    image_part = Part.from_image(Image.load_from_file(image_path))

    # Generate birthday wishes based on the image

    response_stream = model.generate_content(
        [image_part, Part.from_text("Please analyze this bouquet image and create warm birthday wishes in English. Describe what you see and express sincere birthday blessings.")],
        stream=True
    )

    full_response = ""
    for chunk in response_stream:
        if chunk.text:
            print(chunk.text, end="", flush=True)
            full_response += chunk.text
    print("\n")

    print("✅ Birthday wishes generation completed!")
    return full_response
